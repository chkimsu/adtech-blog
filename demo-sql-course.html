SQL 실무 6회

SQLD 로 익힌 문법을 실무 숫자에 맞게 쓰는 법을 익힙니다. 숫자가 틀리는 자리를 먼저 보고, 매일 도는 집계 쿼리를 안전하게 쓰는 데까지 갑니다.

회차주제끝나면 할 수 있는 것
1회처음 보는 테이블 읽기한 행이 무엇인지, 키가 정말 유일한지, NULL 이 어디 있는지 쿼리로 확인합니다
2회NULL 과 집계COUNT 세 가지를 골라 쓰고, NULL 이 집계를 바꾸는 자리를 압니다
3회조인조인 종류를 골라 쓰고, 조인 뒤에 행이 불어났는지 알아채고, 먼저 집계한 뒤 조인합니다
4회CTE 와 윈도우 함수쿼리를 단계로 나누고, 중복 제거와 전날 대비를 윈도우 함수로 씁니다
5회지표와 대시보드날짜 표로 요일과 빈 날을 채우고, 날짜 범위를 바르게 쓰고, 더하면 안 되는 지표를 알아봅니다
6회배치와 적재PRIMARY KEY 와 UNIQUE 로 중복을 막고, 다시 돌려도 같은 결과가 나오게 적재하고, 적재 뒤 검증 쿼리를 씁니다

회차마다 설명 40분, 실습 20분입니다. 화면마다 먼저 던질 질문과 답이 있고, 실습 문제에는 정답 쿼리와 결과가 모두 붙어 있습니다.

준비

내 PC 에 연습 DB 를 만듭니다

MySQL 8.4 하나와 연습 데이터 스크립트 하나면 됩니다. 10분이면 끝납니다.

  1. MySQL 8.4 를 띄웁니다

    Docker 가 있으면 아래 한 줄이면 됩니다. 처음에는 이미지를 받느라 1분쯤 걸리고, 띄운 뒤 20초쯤 기다립니다.

    터미널
    docker run -d --name sql-class -e MYSQL_ROOT_PASSWORD=class -p 3306:3306 mysql:8.4

    Docker 가 없으면 Homebrew 로 설치합니다. brew install mysql@8.4 다음에 brew services start mysql@8.4 를 실행합니다. 이때는 아래 명령의 docker exec -i sql-class mysql -uroot -pclass 자리를 /opt/homebrew/opt/mysql@8.4/bin/mysql -uroot 로 바꿉니다.

  2. 연습 데이터를 넣습니다

    아래 「연습 데이터 스크립트」를 펼쳐 복사 버튼을 누른 뒤, 터미널에서 이 한 줄을 실행합니다. 맥은 복사한 내용을 pbpaste 로 바로 넘길 수 있습니다.

    터미널
    pbpaste | docker exec -i sql-class mysql -uroot -pclass
    연습 데이터 스크립트 (setup.sql)
    setup.sql
    -- SQL 실무 6회 연습 데이터
-- MySQL 8.4 에서 한 번 실행합니다. 다시 실행하면 처음 상태로 돌아갑니다.
-- 시각은 모두 한국 시각입니다.

DROP DATABASE IF EXISTS sql_class;
CREATE DATABASE sql_class DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci;
USE sql_class;

-- 회원. 한 행 = 회원 한 명
CREATE TABLE member (
  member_no   INT          NOT NULL,
  country     CHAR(2)      NULL      COMMENT '가입할 때 고른 국가. 안 고르면 NULL',
  joined_at   DATETIME     NOT NULL  COMMENT '가입 시각',
  status      VARCHAR(10)  NOT NULL  COMMENT 'active 또는 deleted',
  deleted_at  DATETIME     NULL      COMMENT '탈퇴 시각',
  PRIMARY KEY (member_no)
);

-- 모임. 한 행 = 모임 하나
CREATE TABLE club (
  club_no     INT          NOT NULL,
  club_name   VARCHAR(30)  NOT NULL,
  category    VARCHAR(10)  NOT NULL,
  created_at  DATETIME     NOT NULL  COMMENT '만든 시각',
  PRIMARY KEY (club_no)
);

-- 모임 가입. 한 행 = 한 회원이 한 모임에 가입한 것
CREATE TABLE club_member (
  club_no     INT          NOT NULL,
  member_no   INT          NOT NULL,
  role        VARCHAR(10)  NOT NULL  COMMENT 'leader 또는 member',
  joined_at   DATETIME     NOT NULL  COMMENT '가입 시각',
  PRIMARY KEY (club_no, member_no)
);

-- 글. 한 행 = 글 하나
CREATE TABLE post (
  post_no     INT          NOT NULL,
  club_no     INT          NOT NULL,
  member_no   INT          NOT NULL  COMMENT '쓴 회원',
  created_at  DATETIME     NOT NULL  COMMENT '쓴 시각',
  like_count  INT          NULL      COMMENT '좋아요 수. 밤에 집계하기 전이면 NULL',
  PRIMARY KEY (post_no)
);

-- 앱 이벤트 로그. 한 행 = 서버가 받은 기록 한 건
CREATE TABLE event_log (
  log_seq          BIGINT       NOT NULL AUTO_INCREMENT COMMENT '서버가 받은 순서',
  event_id         VARCHAR(12)  NOT NULL  COMMENT '앱이 붙인 이벤트 번호. 앱이 다시 보내면 같은 번호로 또 들어온다',
  member_no        INT          NULL      COMMENT '로그인 전이면 NULL',
  event_type       VARCHAR(20)  NOT NULL  COMMENT 'app_open, post_view, comment_write',
  os               VARCHAR(10)  NOT NULL  COMMENT 'android, ios, web',
  app_version      VARCHAR(10)  NULL      COMMENT 'web 은 NULL',
  event_time       DATETIME     NOT NULL  COMMENT '이벤트가 난 시각',
  received_at      DATETIME     NOT NULL  COMMENT '서버가 받은 시각',
  PRIMARY KEY (log_seq)
);

-- 다른 팀 시스템에서 받은 푸시 발송 기록. 회원 번호를 문자열로 준다
CREATE TABLE push_send (
  push_id     INT          NOT NULL,
  member_no   VARCHAR(20)  NOT NULL  COMMENT '보내는 쪽 시스템이 준 값 그대로',
  sent_at     DATETIME     NOT NULL  COMMENT '보낸 시각',
  PRIMARY KEY (push_id)
);

-- 날짜 표. 한 행 = 하루. 빈 날을 0 으로 채우거나 날짜를 기준으로 조인할 때 쓴다
CREATE TABLE dim_date (
  work_date   DATE         NOT NULL,
  day_name    VARCHAR(3)   NOT NULL  COMMENT '요일',
  is_weekend  TINYINT      NOT NULL  COMMENT '토요일, 일요일이면 1',
  PRIMARY KEY (work_date)
);

-- OS 코드표
CREATE TABLE dim_os (
  os_code     INT          NOT NULL,
  os_name     VARCHAR(10)  NOT NULL,
  PRIMARY KEY (os_code)
);

-- 일별 활동 회원 수. 매일 01시에 도는 배치가 전날 것을 적재한다
CREATE TABLE fact_daily_active (
  work_date   DATE         NOT NULL  COMMENT '대상일',
  country     VARCHAR(3)   NOT NULL  COMMENT '국가. 모르면 ZZ',
  os_code     INT          NOT NULL,
  dau         INT          NOT NULL  COMMENT '그날 이벤트가 하나라도 있는 회원 수',
  updated_at  DATETIME     NOT NULL  COMMENT '적재한 시각',
  PRIMARY KEY (work_date, country, os_code)
);

NULL 은 「값이 없음」이고 「0」이나 빈 글자와 다릅니다. 그래서 등호로는 찾을 수 없습니다.

SELECT COUNT(*) AS member_cnt

2-1

COUNT 는 괄호 안에 따라 다른 것을 셉니다

COUNT(*) 는 행을 셉니다. COUNT(컬럼) 은 NULL 이 아닌 값만 셉니다. 「회원 수」에 맞는 것은 셋 중 하나뿐입니다.

먼저 물어볼 것

  • COUNT(DISTINCT member_no) 도 NULL 은 세지 않습니다. 그래서 로그인 전 기록이 회원 한 명으로 더해지지 않습니다.

평균의 분모는 행 수가 아니라 값이 있는 행 수입니다. like_count 가 NULL 인 글은 분자와 분모에서 함께 빠집니다.

2-4

SELECT 에서 0 으로 나누면 오류 대신 NULL 이 나옵니다

MySQL 은 SELECT 에서 0 으로 나눠도 멈추지 않습니다. NULL 을 내고 경고를 하나 붙입니다. 경고는 결과 끝의 「1 warning」 한 줄로만 보여서 놓치기 쉽습니다.

NULL 이 섞인 표를 집계할 때 쓰는 다섯 가지

틀
-- 1. 사람 수는 DISTINCT 로 센다. NULL 은 세지 않는다
SELECT COUNT(DISTINCT 회원_컬럼) FROM 테이블;

-- 2. 평균을 내기 전에 NULL 을 뺄지 0 으로 볼지 정한다
SELECT AVG(컬럼), AVG(COALESCE(컬럼, 0)) FROM 테이블;

-- 3. 없는 것을 찾을 때는 NOT EXISTS
SELECT a.키 FROM 테이블_a AS a
WHERE NOT EXISTS (SELECT 1 FROM 테이블_b AS b WHERE b.키 = a.키);

-- 4. 분모는 NULLIF 로 감싼다
SELECT 분자 / NULLIF(분모, 0) FROM 테이블;

-- 5. 조건부 집계. COUNT 안의 CASE 에는 ELSE 를 쓰지 않는다
SELECT 키, SUM(조건), COUNT(DISTINCT CASE WHEN 조건 THEN id_컬럼 END)
FROM 테이블 GROUP BY 키 ORDER BY 키;

4-1

4-2

중복 이벤트는 ROW_NUMBER 로 한 행만 남깁니다

이벤트 수만 셀 때는 COUNT(DISTINCT event_id) 로 됩니다. 행을 남겨 다른 집계에 쓰려면 이벤트마다 한 행만 골라야 합니다.

먼저 물어볼 것

SELECT DISTINCT * 로 중복을 지우면 몇 행이 남을까요?

답

66행입니다. 원래 행 수 그대로입니다. 다시 들어온 행은 log_seq 와 received_at 이 다릅니다. 그래서 DISTINCT 는 다른 행으로 봅니다.

4-3

최신 값은 MAX 가 아니라 최신 행에서 꺼냅니다

MAX 는 컬럼마다 따로 가장 큰 값을 찾습니다. 가장 늦은 행의 값을 가져오는 함수가 아닙니다.

먼저 물어볼 것

회원 105 의 마지막 app_version 을 MAX(app_version) 으로 구하면 무엇이 나올까요?

답

9.8.0 이 나옵니다. 마지막 이벤트의 버전은 10.1.0 입니다. app_version 은 문자열입니다. 문자열은 앞 글자부터 비교합니다. 첫 글자 '9' 가 '1' 보다 큽니다. 그래서 '9.8.0' 이 가장 큰 값입니다.

앱을 업데이트한 회원 105, 106 으로 비교합니다.

일별 표에는 날짜 옆에 요일을 같이 둡니다. 주말에 숫자가 어떻게 바뀌는지 바로 보이기 때문입니다. 요일과 주말 여부는 날짜 표 dim_date 에 하루 한 행씩 적혀 있습니다.

먼저 물어볼 것

10월 1일부터 7일까지 평일의 하루 평균 DAU 와 주말의 하루 평균 DAU 는 각각 몇 명일까요?

답

평일은 5.4명, 주말은 4.5명입니다. 이 주의 주말은 10월 3일 토요일과 4일 일요일입니다. 4일의 DAU 가 3명으로 이 주에서 가장 적습니다. 주말에 든 날은 2개뿐입니다. 그래서 4일 하루가 주말 평균을 크게 낮춥니다.

5-2

날짜 범위는 반열린 구간으로 씁니다

DATETIME 컬럼과 비교할 때 날짜만 쓰면 그날 0시 0분 0초가 됩니다. BETWEEN 의 끝을 날짜로 쓰면 마지막 날은 0시 정각의 행만 들어갑니다.

먼저 물어볼 것

아래 두 쿼리는 10월 1일부터 3일까지 쓴 글을 셉니다. 각각 몇 개가 나올까요?

답

5-3

빈 날짜는 날짜 표를 왼쪽에 두고 0 으로 채웁니다

GROUP BY 는 행이 있는 날만 냅니다. 글이 하나도 없는 날은 0 이 아니라 행 자체가 없습니다.

5-5

비율은 합을 합으로 나눕니다

글당 좋아요처럼 나눠서 만든 지표는 평균을 다시 평균하면 틀립니다. 분자와 분모를 각각 더한 뒤에 나눕니다.

5-6

합계 행의 NULL 은 GROUPING() 으로 구분합니다

GROUP BY 끝에 WITH ROLLUP 을 쓰면 결과 아래에 소계와 합계 행이 더 나옵니다. 그 행의 묶음 컬럼은 NULL 로 나옵니다.

왼쪽 결과에는 country 가 NULL 인 행이 2개 있습니다. 각각 무엇일까요?

대시보드 일별 표

10월 1일부터 7일까지 대시보드에 올릴 일별 표를 만듭니다. 칸은 날짜, 요일, DAU, 글 수, 글 쓴 회원 수, 글 쓴 회원 1인당 글 수입니다. 글이 없는 날도 한 행이 나와야 합니다. 1인당 글 수는 나눌 회원이 없으면 NULL 로 둡니다.

힌트
  • DAU 와 글 집계를 CTE 로 따로 만듭니다. 그다음 dim_date 에 하나씩 LEFT JOIN 합니다. event_log 와 post 를 먼저 조인하면 행이 곱해집니다(3회).
  • DAU 는 event_log 만으로 셉니다. 날짜는 DATE(event_time) 로 만듭니다. 범위는 반열린 구간으로 겁니다.
  • 0 으로 나누지 않게 NULLIF(writer_cnt, 0) 으로 나눕니다(2회).
정답

  SELECT DATE(event_time)          AS work_date,
         COUNT(DISTINCT member_no) AS dau

  GROUP BY work_date
),
posts AS (

         COUNT(DISTINCT member_no) AS writer_cnt

정리

대시보드 일별 표의 틀

틀
-- 1. 지표마다 따로 집계
--    날짜: DATE(시각)
--    범위: 시작일을 넣고 끝일의 다음 날을 빼는 반열린 구간
WITH metric AS (
  SELECT DATE(시각)                AS work_date,
         COUNT(DISTINCT member_no) AS dau,
         SUM(분자)                 AS num,
         SUM(분모)                 AS den
  FROM 테이블
  WHERE 시각 >= '시작일'
    AND 시각 <  '끝일의 다음 날'
  GROUP BY work_date
)
-- 2. 날짜 표를 왼쪽에 두고 LEFT JOIN
--    빈 날도 한 행, 요일과 주말도 날짜 표에서
--    세는 지표: COALESCE 로 0
--    비율: 합을 합으로 나누고, 분모가 0 이면 NULL
SELECT d.work_date,
       d.day_name,
       COALESCE(m.dau, 0)       AS dau,
       m.num / NULLIF(m.den, 0) AS ratio
FROM dim_date AS d
LEFT JOIN metric AS m ON m.work_date = d.work_date
WHERE d.work_date >= '시작일'
  AND d.work_date <  '끝일의 다음 날'
ORDER BY d.work_date;

-- 3. 주간 순사용자 수: 일별 dau 의 합이 아니라 원천에서 COUNT(DISTINCT)

6-2

키가 없는 표에 두 번 넣으면 두 배가 됩니다

배치는 생각보다 자주 다시 돕니다. 실패한 날을 다시 돌리고, 규칙을 고친 뒤에도 다시 돌립니다. 키가 없는 표는 두 번째 실행을 막지 못합니다.

먼저 물어볼 것

키가 없는 표에 같은 INSERT 를 두 번 돌리면 어떻게 될까요?

답

WHERE c.키 IS NULL;
